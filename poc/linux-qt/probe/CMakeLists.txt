cmake_minimum_required(VERSION 3.21)
project(qt-probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Network Widgets WebEngineWidgets)
qt_standard_project_setup()
qt_add_executable(qt-probe main.cpp)
target_link_libraries(qt-probe PRIVATE Qt6::Network Qt6::Widgets Qt6::WebEngineWidgets)
install(TARGETS qt-probe RUNTIME DESTINATION bin)
